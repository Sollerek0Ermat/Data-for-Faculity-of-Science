/** * 🔑 Set / Reset Password page * Supports: * - First time set password (normal flow) * - Password reset via Microsoft OAuth (with ?token=xxx) */ document.addEventListener('DOMContentLoaded', () => { renderTopbar(); // تحقق هل جايين من Microsoft reset const urlParams = new URLSearchParams(location.search); const resetToken = urlParams.get('token'); const errorMsg = urlParams.get('error'); // لو فيه error من الباك if (errorMsg) { const msg = document.getElementById('msg'); if (msg) { msg.textContent = '❌ ' + decodeURIComponent(errorMsg); msg.style.color = 'var(--danger)'; } // نمنع الإرسال const form = document.getElementById('setPwdForm'); if (form) form.style.display = 'none'; setTimeout(() => { location.href = 'admin-login.html'; }, 4000); return; } // لو جاي من reset، غير العنوان if (resetToken) { const h1 = document.querySelector('.login-card h1'); const p = document.querySelector('.login-card > p'); if (h1) h1.textContent = 'Reset Password'; if (p) p.textContent = 'Enter your new password'; } else { // وضع "First time" — لازم يكون مسجل دخول if (!Auth.isLoggedIn()) { window.location.href = 'login.html'; return; } } const form = document.getElementById('setPwdForm'); const btn = document.getElementById('submitBtn'); const msg = document.getElementById('msg'); form.addEventListener('submit', async (e) => { e.preventDefault(); const newPwd = document.getElementById('newPwd').value; const confirmPwd = document.getElementById('confirmPwd').value; msg.textContent = ''; if (newPwd !== confirmPwd) { msg.textContent = '❌ Passwords do not match'; msg.style.color = 'var(--danger)'; return; } if (newPwd.length < 8) { msg.textContent = '❌ Password too short (min 8)'; msg.style.color = 'var(--danger)'; return; } btn.disabled = true; msg.textContent = 'Saving...'; msg.style.color = 'var(--muted)'; try { if (resetToken) { // وضع reset password عبر Microsoft OAuth await API.resetPassword({ token: resetToken, newPassword: newPwd }); msg.textContent = '✅ Password reset successfully'; } else { // وضع first time set password await API.adminLogin({ email: Auth.getUser()?.email || '', password: newPwd }); msg.textContent = '✅ Password saved'; } msg.style.color = 'var(--success)'; UI.success('Password saved'); setTimeout(() => { window.location.href = 'admin.html'; }, 1000); } catch (err) { console.error(err); msg.textContent = '❌ ' + (err.message || 'Failed'); msg.style.color = 'var(--danger)'; btn.disabled = false; } }); });
